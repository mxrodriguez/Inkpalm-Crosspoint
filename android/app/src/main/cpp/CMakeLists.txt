cmake_minimum_required(VERSION 3.18)
project(crosspoint-android)

# Resolve to absolute real path (resolves symlinks, avoids /mnt/d/ issues)
get_filename_component(CROSSPOINT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../" REALPATH)
set(STUB_DIR "${CMAKE_CURRENT_SOURCE_DIR}/android_stubs")

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

add_definitions(-DANDROID=1)
add_definitions(-DCROSPOINT_ANDROID=1)

include_directories(
    ${CMAKE_SOURCE_DIR}
    ${STUB_DIR}
    ${CROSSPOINT_ROOT}/lib
    ${CROSSPOINT_ROOT}/lib/ZipFile
    ${CROSSPOINT_ROOT}/lib/uzlib/src
    ${CROSSPOINT_ROOT}/lib/expat
    ${CROSSPOINT_ROOT}/lib/EpdFont
    ${CROSSPOINT_ROOT}/lib/GfxRenderer
)

# uzlib (compression) - C library
add_library(uzlib STATIC "${CROSSPOINT_ROOT}/lib/uzlib/src/tinflate.c" "${STUB_DIR}/uzlib_checksums.c")

# Logging (Android stub)
add_library(crosspoint_logging STATIC "${STUB_DIR}/Logging.cpp")

# Arduino compatibility (Print::printf, Serial, millis, etc.)
add_library(arduino_compat STATIC "${STUB_DIR}/ArduinoCompat.cpp")

# Utf8
add_library(utf8 STATIC "${CROSSPOINT_ROOT}/lib/Utf8/Utf8.cpp")

# InflateReader (stub impl in android_stubs)
add_library(inflateReader STATIC "${STUB_DIR}/InflateReader.cpp")
target_link_libraries(inflateReader uzlib crosspoint_logging)

# ZipFile
add_library(zipfile STATIC "${CROSSPOINT_ROOT}/lib/ZipFile/ZipFile.cpp")
target_link_libraries(zipfile inflateReader crosspoint_logging)

# Expat (XML parser)
add_library(expat STATIC
    "${CROSSPOINT_ROOT}/lib/expat/xmlparse.c"
    "${CROSSPOINT_ROOT}/lib/expat/xmlrole.c"
    "${CROSSPOINT_ROOT}/lib/expat/xmltok.c"
)
target_compile_definitions(expat PRIVATE
    XML_GE=1
    XML_CONTEXT_BYTES=1024
    XML_DTD=1
    XML_NS=1
)

# JsonParser
add_library(jsonparser STATIC
    "${CROSSPOINT_ROOT}/lib/JsonParser/ReleaseJsonParser.cpp"
    "${CROSSPOINT_ROOT}/lib/JsonParser/StreamingJsonParser.cpp"
)

# EpdFont — real font engine (EpdFont.cpp, EpdFontFamily.cpp, FontDecompressor.cpp)
add_library(epdfont STATIC
    "${CROSSPOINT_ROOT}/lib/EpdFont/EpdFont.cpp"
    "${CROSSPOINT_ROOT}/lib/EpdFont/EpdFontFamily.cpp"
    "${CROSSPOINT_ROOT}/lib/EpdFont/FontDecompressor.cpp"
)
target_link_libraries(epdfont inflateReader crosspoint_logging)
target_include_directories(epdfont PUBLIC
    "${CROSSPOINT_ROOT}/lib/EpdFont"
    "${STUB_DIR}"
    "${CROSSPOINT_ROOT}/lib"
)

# GfxRenderer — real rendering engine
add_library(gfxrenderer STATIC
    "${CROSSPOINT_ROOT}/lib/GfxRenderer/GfxRenderer.cpp"
    "${CROSSPOINT_ROOT}/lib/GfxRenderer/Bitmap.cpp"
    "${CROSSPOINT_ROOT}/lib/GfxRenderer/BitmapHelpers.cpp"
    "${CROSSPOINT_ROOT}/lib/GfxRenderer/FontCacheManager.cpp"
)
target_link_libraries(gfxrenderer epdfont utf8 crosspoint_logging arduino_compat)
target_include_directories(gfxrenderer PUBLIC
    "${CROSSPOINT_ROOT}/lib/GfxRenderer"
    "${STUB_DIR}"
    "${CROSSPOINT_ROOT}/lib"
)

# Serialization
add_library(serialization STATIC
    "${CROSSPOINT_ROOT}/lib/Serialization/ObfuscationUtils.cpp"
)
target_link_libraries(serialization crosspoint_logging)
target_include_directories(serialization PUBLIC
    "${CROSSPOINT_ROOT}/lib/Serialization"
    "${STUB_DIR}"
    "${CROSSPOINT_ROOT}/lib"
)

# I18n — needs generated I18nKeys.h/I18nStrings.h from YAML translations
# These are generated by PlatformIO build; create minimal stubs for Android
# For now, skip I18n compilation — it's only needed for UI activities, not core rendering

# XmlParserUtils — header-only utilities for expat
add_library(xmlparserutils INTERFACE)
target_include_directories(xmlparserutils INTERFACE
    "${CROSSPOINT_ROOT}/lib/XmlParserUtils"
    "${CROSSPOINT_ROOT}/lib/expat"
    "${STUB_DIR}"
    "${CROSSPOINT_ROOT}/lib"
)
target_link_libraries(xmlparserutils INTERFACE expat)

# JpegToBmpConverter (Android stub — real file uses JPEGDEC which isn't available)
add_library(jpegtobmp STATIC
    "${STUB_DIR}/JpegToBmpConverter.cpp"
)
target_link_libraries(jpegtobmp crosspoint_logging)
target_include_directories(jpegtobmp PUBLIC
    "${CROSSPOINT_ROOT}/lib/JpegToBmpConverter"
    "${STUB_DIR}"
    "${CROSSPOINT_ROOT}/lib"
)

# PngToBmpConverter (Android stub — real file's own PNG parsing needs full InflateReader)
add_library(pngtobmp STATIC
    "${STUB_DIR}/PngToBmpConverter.cpp"
)
target_link_libraries(pngtobmp inflateReader crosspoint_logging)
target_include_directories(pngtobmp PUBLIC
    "${CROSSPOINT_ROOT}/lib/PngToBmpConverter"
    "${STUB_DIR}"
    "${CROSSPOINT_ROOT}/lib"
)

# Epub — core book parsing engine
add_library(epub STATIC

    "${CROSSPOINT_ROOT}/lib/Epub/Epub/converters/ImageDecoderFactory.cpp"
    "${CROSSPOINT_ROOT}/lib/Epub/Epub/converters/ImageToFramebufferDecoder.cpp"

    zipfile inflateReader expat xmlparserutils

    jpegtobmp pngtobmp

target_include_directories(epub PUBLIC
    "${CROSSPOINT_ROOT}/lib/Epub"
    "${CROSSPOINT_ROOT}/lib/Epub/Epub"
    "${CROSSPOINT_ROOT}/lib/Epub/Epub/blocks"
    "${CROSSPOINT_ROOT}/lib/Epub/Epub/converters"
    "${CROSSPOINT_ROOT}/lib/Epub/Epub/css"
    "${CROSSPOINT_ROOT}/lib/Epub/Epub/hyphenation"
    "${CROSSPOINT_ROOT}/lib/Epub/Epub/parsers"
    "${STUB_DIR}"
    "${CROSSPOINT_ROOT}/lib"
    "${CROSSPOINT_ROOT}/lib/ZipFile"
    "${CROSSPOINT_ROOT}/lib/expat"

    "${CROSSPOINT_ROOT}/lib/EpdFont"

    "${CROSSPOINT_ROOT}/lib/Serialization"

)
